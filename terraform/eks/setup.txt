# ------------------------------------------------------------
# EKS Terraform Setup (vLLM + optional HAMI GPU node group)
# ------------------------------------------------------------

# 1) Install prerequisites (Linux example)
# Terraform: https://developer.hashicorp.com/terraform/install
# kubectl:   https://kubernetes.io/docs/tasks/tools/
# AWS CLI v2 install guide:
# https://docs.aws.amazon.com/cli/latest/userguide/getting-started-install.html

# 2) Configure AWS credentials (choose one)
aws configure
# or for SSO profiles:
# aws configure sso

# 3) Verify caller identity and default region
aws sts get-caller-identity
aws configure get region

# Optional: override region in the shell for this session
export AWS_REGION=us-west-2

# ------------------------------------------------------------
# GPU quota preparation for NVIDIA L4 Spot (g6 family)
# ------------------------------------------------------------
# L4 on AWS is typically delivered via g6 instance types.
# For Spot-backed GPU workers, request EC2 quota increase for G/VT Spot usage.

# Inspect relevant EC2 quota names and current limits:
aws service-quotas list-service-quotas \
  --service-code ec2 \
  --region ${AWS_REGION:-us-west-2} \
  --query 'Quotas[?contains(QuotaName, `Spot`) || contains(QuotaName, `G and VT`) || contains(QuotaName, `Running On-Demand G`)][QuotaName,QuotaCode,Value]' \
  --output table

# Request increase for "All G and VT Spot Instance Requests" quota.
# This request is account/region specific.
SPOT_QUOTA_CODE=$(aws service-quotas list-service-quotas \
  --service-code ec2 \
  --region ${AWS_REGION:-us-west-2} \
  --query 'Quotas[?contains(QuotaName, `All G and VT Spot Instance Requests`) == `true`].QuotaCode | [0]' \
  --output text)

# Example request (set desired limit based on your target capacity):
aws service-quotas request-service-quota-increase \
  --service-code ec2 \
  --quota-code "$SPOT_QUOTA_CODE" \
  --desired-value 16 \
  --region ${AWS_REGION:-us-west-2}

# If your workload can fall back to On-Demand g6, request that quota too.

# ------------------------------------------------------------
# Terraform workflow
# ------------------------------------------------------------

# 4) Copy variables and edit values
cp terraform/eks/terraform.tfvars.example terraform/eks/terraform.tfvars

# 5) Init / Plan / Apply
terraform -chdir=terraform/eks init -backend=false -input=false
terraform -chdir=terraform/eks plan --var-file=terraform.tfvars -lock=false
terraform -chdir=terraform/eks apply --var-file=terraform.tfvars -lock=false

# 6) Configure kubectl for the new EKS cluster
aws eks update-kubeconfig \
  --name vllm-ephemeral-eks \
  --region ${AWS_REGION:-us-west-2}

# 7) Verify nodes and deploy workloads
kubectl get nodes -o wide

# If needed, enforce gpu=on label manually:
# kubectl label node -l eks.amazonaws.com/nodegroup=vllm-ephemeral-eks-hami-gpu gpu=on --overwrite

# Deploy manifests (base and optional HAMI GPU overlay)
kubectl apply -k k8s/base
# kubectl apply -k k8s/overlays/hami-gpu

# ------------------------------------------------------------
# Cleanup
# ------------------------------------------------------------
terraform -chdir=terraform/eks destroy --var-file=terraform.tfvars -lock=false
